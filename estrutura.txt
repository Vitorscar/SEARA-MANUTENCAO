function fmtMin(ms){
  const min = Math.max(0, Math.floor(ms/60000));
  const h = Math.floor(min/60), m = min%60;
  return h > 0 ? `${h}h${String(m).padStart(2,'0')}` : `${m}min`;
}
function fmtDuracaoMin(min){
  const h = Math.floor(min/60), m = min%60;
  return h > 0 ? `${h}h ${m}min` : `${m}min`;
}
function minutosDesde(iso){ return Math.max(0, Math.round((Date.now()-iso)/60000)); }
function escapeHtml(s){ return String(s??'').replace(/[&<>"']/g, c=>({'&':'&amp;','<':'&lt;','>':'&gt;','"':'&quot;',"'":'&#39;'}[c])); }
function nowStr(){
  const d = new Date();
  return d.toLocaleTimeString('pt-BR',{hour:'2-digit',minute:'2-digit'});
}
function turnoPorHora(h){
  if(h>=6 && h<14) return '1º Turno';
  if(h>=14 && h<22) return '2º Turno';
  return '3º Turno';
}
function turnoAtual(){ return turnoPorHora(new Date().getHours()); }
function getMaquina(id){ return db.maquinas.find(m=>m.id===id); }
function getTecnico(id){ return db.tecnicos.find(t=>t.id===id); }

function toast(msg, tipo){
  const el = document.createElement('div');
  el.className = 'toast' + (tipo ? ' '+tipo : '');
  el.textContent = msg;
  document.getElementById('toastwrap').appendChild(el);
  setTimeout(()=>el.remove(), 3200);
}

/* Undo */
function registrarDesfazer(texto, fn){
  ultimaAcao = fn;
  document.getElementById('undoText').textContent = texto;
  document.getElementById('undoBar').classList.add('show');
  clearTimeout(undoTimer);
  undoTimer = setTimeout(()=>{
    document.getElementById('undoBar').classList.remove('show');
    ultimaAcao = null;
  }, 6000);
}
function desfazer(){
  if(ultimaAcao) ultimaAcao();
  document.getElementById('undoBar').classList.remove('show');
  ultimaAcao = null;
  clearTimeout(undoTimer);
  salvarDB();
  toast('Ação desfeita');
  render();
}

/* Alerta crítico */
function alertaCritico(){
  document.body.classList.add('critical-flash');
  setTimeout(()=>document.body.classList.remove('critical-flash'), 2000);
  if(navigator.vibrate) navigator.vibrate([200,100,200,100,250]);
  try{
    const ctx = new (window.AudioContext||window.webkitAudioContext)();
    [0,260,520].forEach(delay=>{
      setTimeout(()=>{
        const osc = ctx.createOscillator();
        const gain = ctx.createGain();
        osc.type='square'; osc.frequency.value=880;
        gain.gain.setValueAtTime(.14, ctx.currentTime);
        gain.gain.exponentialRampToValueAtTime(.001, ctx.currentTime+0.16);
        osc.connect(gain); gain.connect(ctx.destination);
        osc.start(); osc.stop(ctx.currentTime+0.18);
      }, delay);
    });
  }catch(e){}
}

function atualizarBadges(){
  const abertas = db.paradas.filter(p=>p.status!=='encerrada').length;
  ['navRadarCount','navRadarCount2'].forEach(id=>{
    const el = document.getElementById(id);

  });

function irParaRegistro(){
  draft = {categoria:null, subcausa:null, acaoComponente:null, acaoPreventiva:null, impacto:'Alto', foto:null, paradaEditando:null};

function nivel(p){
  const m = minutosDesde(p.horaInicio);
  if(m >= 60) return 'crit';
  if(m >= 30) return 'warn';
  return 'ok';
}

function paradaCardHTML(p){
  const m = getMaquina(p.maquinaId);
  const t = p.tecnicoId ? getTecnico(p.tecnicoId) : null;
  const niv = nivel(p);
  const cls = p.status === 'atendendo' ? 'atendendo' : niv === 'crit' ? 'crit' : niv === 'warn' ? 'warn' : '';
  const statusLbl = p.status === 'atendendo' ? `<b>Em atendimento${t?' · '+escapeHtml(t.nome):''}</b>` : '<b>Aguardando técnico</b>';
  return `
    <div class="card ${cls}">
      <div class="top">
        <div style="display:flex; gap:10px; align-items:flex-start; min-width:0; flex:1;">
          ${p.foto ? `<img class="thumb" src="${p.foto}" alt="">` : ''}
          <div style="min-width:0;">
            <div class="title">#${p.numero} · ${escapeHtml(m?.nome || '—')}</div>
            <div class="subtitle">${escapeHtml(p.setor)} · ${escapeHtml(p.turno)} · ${escapeHtml(p.categoria||'Sem categoria')}</div>
          </div>
        </div>
        <div class="tempo"><span class="lbl">parada há</span>${fmtMin(Date.now()-p.horaInicio)}</div>
      </div>
      <div class="meta">
        <span>${statusLbl}</span>
        <span>Impacto: <b style="color:${impactoCor(p.impacto)}">${p.impacto}</b></span>
      </div>
      <div class="actions">
        ${p.status==='aguardando' ? `<button class="btn sm sun" data-act="atender" data-id="${p.id}">Assumir</button>`:''}
        <button class="btn sm primary" data-act="encerrar" data-id="${p.id}">Encerrar</button>
        <button class="btn sm ghost" data-act="editar" data-id="${p.id}">Corrigir</button>
      </div>
    </div>
  `;
}
function impactoCor(i){
  return {'Baixo':'#1E7A3D','Médio':'#D89A00','Alto':'#C77700','Crítico':'#C8102E'}[i] || '#6B6460';
}
function bindParadaActions(){
  document.querySelectorAll('[data-act]').forEach(btn=>{
    btn.onclick = ()=>{
      const p = db.paradas.find(x=>x.id === btn.dataset.id);
      if(!p) return;
      const act = btn.dataset.act;
      if(act === 'atender') abrirAssumir(p);
      if(act === 'encerrar') abrirEncerrar(p);
      if(act === 'editar') abrirCorrigir(p);
    };
  });
}

function wireRegistro(){
  // Categoria → subcausa
  const grid = document.getElementById('catGrid');
  grid.addEventListener('click', e=>{
    const btn = e.target.closest('.cat-btn'); if(!btn) return;
    [...grid.children].forEach(b=>b.classList.remove('active'));
    btn.classList.add('active');
    draft.categoria = btn.dataset.cat;
    draft.subcausa = null;
    renderSubcausasRegistro();
    document.getElementById('subcausaWrap').classList.remove('hidden');
  });
  // Se categoria já vem do draft/parada, mostrar subcausas
  const catInicial = draft.paradaEditando
    ? db.paradas.find(x=>x.id===draft.paradaEditando)?.categoria
    : draft.categoria;
  if(catInicial){
    draft.categoria = catInicial;
    draft.subcausa = draft.paradaEditando
      ? db.paradas.find(x=>x.id===draft.paradaEditando)?.subcausa
      : draft.subcausa;
    renderSubcausasRegistro();
    document.getElementById('subcausaWrap').classList.remove('hidden');
  }

  // Radio: ação componente / preventiva
  wireRadioRegistro('acaoCompGrid', v=>draft.acaoComponente = v);
  wireRadioRegistro('acaoPrevGrid', v=>draft.acaoPreventiva = v);

  // Impacto
  document.getElementById('fImpacto').onchange = e=>{
    draft.impacto = e.target.value;
    if(draft.impacto === 'Crítico'){
      if(navigator.vibrate) navigator.vibrate(80);
    }
  };
}

function renderSubcausasRegistro(){
  const cont = document.getElementById('subcausas');
  if(!cont) return;
  cont.innerHTML = '';
  if(!draft.categoria) return;
  SUBCAUSAS[draft.categoria].forEach(s=>{
    const c = document.createElement('div');
    c.className = 'chip-opt' + (draft.subcausa === s ? ' active' : '');
    c.textContent = s;
    c.onclick = ()=>{
      [...cont.children].forEach(x=>x.classList.remove('active'));
      c.classList.add('active');
      draft.subcausa = s;
    };
    cont.appendChild(c);
  });
}

function wireRadioRegistro(id, setter){
  const el = document.getElementById(id);
  el.addEventListener('click', e=>{
    const c = e.target.closest('.radio-opt'); if(!c) return;
    [...el.children].forEach(x=>x.classList.remove('active'));
    c.classList.add('active');
    setter(c.dataset.val);
  });
}

function cancelarRegistro(){
  draft = {categoria:null, subcausa:null, acaoComponente:null, acaoPreventiva:null, impacto:'Alto', foto:null, paradaEditando:null};
  navigate('radar');
}

function processarFoto(e){
  const file = e.target.files[0]; if(!file) return;
  const reader = new FileReader();
  reader.onload = ()=>{
    draft.foto = reader.result;
    document.getElementById('photoImg').src = draft.foto;
    document.getElementById('photoPreview').style.display = 'inline-block';
    if(navigator.vibrate) navigator.vibrate(20);
  };
  reader.readAsDataURL(file);
}
function removerFoto(){
  draft.foto = null;
  const inp = document.getElementById('fFotoInput'); if(inp) inp.value = '';
  const prev = document.getElementById('photoPreview'); if(prev) prev.style.display = 'none';
}

/* ---------- Salvar registro ---------- */
function salvarRegistro(){
  const maquinaNome = document.getElementById('fMaquina').value.trim();
  const setor = document.getElementById('fSetor').value;
  const turno = document.getElementById('fTurno').value;
  const duracao = document.getElementById('fDuracao').value;
  const impacto = document.getElementById('fImpacto').value;
  const componente = document.getElementById('fComponente').value.trim();
  const causaRaiz = document.getElementById('fCausaRaiz').value.trim();
  const responsavel = document.getElementById('fResponsavel').value.trim();
  const observacao = document.getElementById('fObservacao').value.trim();

  if(!maquinaNome){ toast('Informe a máquina.','red'); return; }
  if(!setor){ toast('Selecione o setor.','red'); return; }
  if(!draft.categoria){ toast('Selecione o tipo de falha.','red'); return; }
  if(!draft.acaoComponente){ toast('Selecione a ação no componente.','red'); return; }
  if(!responsavel){ toast('Informe o responsável.','red'); return; }

  lastSetor = setor;
  localStorage.setItem('ultimoSetor', setor);
  localStorage.setItem('tecnicoNome', responsavel);

  // Cria ou busca máquina pelo nome
  let maq = db.maquinas.find(m=>m.nome.toLowerCase() === maquinaNome.toLowerCase());
  if(!maq){
    maq = {id:'m'+Date.now(), nome:maquinaNome, setor, prioridade:2, status:'parada', qr:''};
    db.maquinas.push(maq);
  }

  if(draft.paradaEditando){
    const p = db.paradas.find(x=>x.id === draft.paradaEditando);
    if(p){
      p.setor = setor; p.turno = turno;
      p.duracaoMin = duracao ? parseInt(duracao,10) : p.duracaoMin;
      p.impacto = impacto;
      p.categoria = draft.categoria;
      p.subcausa = draft.subcausa || '—';
      p.componente = componente;
      p.causaRaiz = causaRaiz;
      p.acaoComponente = draft.acaoComponente;
      p.acaoPreventiva = draft.acaoPreventiva || '—';
      p.responsavel = responsavel;
      p.observacao = observacao;
      p.foto = draft.foto;
      toast('Registro #'+p.numero+' atualizado','green');
    }
  } else {
    const novo = {
      id:'p'+Date.now(), numero:db.seqParada++,
      maquinaId: maq.id, setor, turno,
      horaInicio: Date.now(), horaFim: null, duracaoMin: duracao ? parseInt(duracao,10) : null,
      impacto, status:'aguardando', tecnicoId:null,
      categoria: draft.categoria, subcausa: draft.subcausa || '—',
      componente, causaRaiz,
      acaoComponente: draft.acaoComponente, acaoPreventiva: draft.acaoPreventiva || '—',
      responsavel, observacao, foto: draft.foto
    };
    db.paradas.unshift(novo);
    maq.status = 'parada';

    // Cria OS correspondente
    db.os.unshift({
      id:'os'+Date.now(), paradaId: novo.id, maquinaId: maq.id,
      titulo: causaRaiz || draft.categoria, coluna:'fila', tecnicoId:null
    });

    if(impacto === 'Crítico') alertaCritico();
    toast(`Parada #${novo.numero} registrada · ${maquinaNome}`, impacto === 'Crítico' ? 'red' : 'green');

    // Undo
    const idNovo = novo.id;
    registrarDesfazer(`Parada #${novo.numero} registrada`, ()=>{
      db.paradas = db.paradas.filter(p=>p.id !== idNovo);
      db.os = db.os.filter(o=>o.paradaId !== idNovo);
      const m = getMaquina(maq.id); if(m) m.status = 'ok';
    });
  }

  salvarDB();
  draft = {categoria:null, subcausa:null, acaoComponente:null, acaoPreventiva:null, impacto:'Alto', foto:null, paradaEditando:null};
  navigate('radar');
}

const COLUNAS = [
  {id:'fila', nome:'A Fazer'},
  {id:'andamento', nome:'Em Andamento'},
  {id:'aguardando', nome:'Aguardando'},
  {id:'concluido', nome:'Concluído'}
];
function renderOS(){
  document.getElementById('view').innerHTML = `
    <div class="section-head"><h2>Ordens de serviço</h2></div>
    <div class="kanban" id="kanban"></div>
  `;
  const kanban = document.getElementById('kanban');
  kanban.innerHTML = COLUNAS.map(c=>{
    const items = db.os.filter(o=>o.coluna === c.id);
    return `
      <div class="kcol" data-col="${c.id}">
        <h3><span>${c.nome}</span><span>${items.length}</span></h3>
        ${items.map(o=>{
          const m = getMaquina(o.maquinaId);
          const t = o.tecnicoId ? getTecnico(o.tecnicoId) : null;
          const cls = o.coluna === 'andamento' ? 'sun' : o.coluna === 'concluido' ? 'done' : '';
          return `
            <div class="kcard ${cls}" draggable="true" data-id="${o.id}">
              <div class="num">#${o.id.slice(-4)}</div>
              <div class="title">${escapeHtml(o.titulo)}</div>
              <div class="sub">${m ? escapeHtml(m.nome) : ''}${t ? ' · '+escapeHtml(t.nome) : ''}</div>
            </div>
          `;
        }).join('') || '<div class="empty" style="padding:14px; font-size:12px;">vazio</div>'}
      </div>
    `;
  }).join('');

  // Drag & drop
  kanban.querySelectorAll('.kcard').forEach(card=>{
    card.addEventListener('dragstart', e=>{
      draggedOSId = card.dataset.id;
      e.dataTransfer.effectAllowed = 'move';
    });
  });
  kanban.querySelectorAll('.kcol').forEach(col=>{
    col.addEventListener('dragover', e=>{ e.preventDefault(); col.classList.add('dragover'); });
    col.addEventListener('dragleave', ()=> col.classList.remove('dragover'));
    col.addEventListener('drop', e=>{
      e.preventDefault();
      col.classList.remove('dragover');
      const os = db.os.find(o=>o.id === draggedOSId);
      if(!os) return;
      os.coluna = col.dataset.col;
      const p = db.paradas.find(p=>p.id === os.paradaId);
      if(p){
        if(col.dataset.col === 'andamento') p.status = 'atendendo';
        if(col.dataset.col === 'concluido' && p.status !== 'encerrada'){
          p.status = 'encerrada';
          p.horaFim = Date.now();
          p.duracaoMin = Math.round((p.horaFim - p.horaInicio)/60000);
          const m = getMaquina(p.maquinaId); if(m) m.status = 'ok';
        }
      }
      salvarDB(); render();
    });
  });
}

function renderRelatorios(){
  const encerradas = db.paradas.filter(p=>p.status === 'encerrada');
  const totalMin = encerradas.reduce((s,p)=>s+(p.duracaoMin||0),0);

  // Pareto por categoria
  const porCat = {};
  encerradas.forEach(p=>{
    const c = p.categoria || 'Sem categoria';
    porCat[c] = (porCat[c]||0) + (p.duracaoMin||0);
  });
  const maxCat = Math.max(...Object.values(porCat), 1);
  const ordenado = Object.entries(porCat).sort((a,b)=>b[1]-a[1]);

  // Top máquinas
  const porMaq = {};
  encerradas.forEach(p=>{
    const m = getMaquina(p.maquinaId);
    const n = m?.nome || '—';
    porMaq[n] = (porMaq[n]||0) + 1;
  });
  const topMaq = Object.entries(porMaq).sort((a,b)=>b[1]-a[1]).slice(0,5);

  document.getElementById('view').innerHTML = `
    <div class="section-head"><h2>Indicadores — geral</h2></div>
    <div class="kpi-grid">
      <div class="kpi"><b>${encerradas.length}</b><span>Paradas encerradas</span></div>
      <div class="kpi"><b>${mttr ? mttr+'m' : '—'}</b><span>MTTR médio</span></div>
      <div class="kpi"><b>${fmtDuracaoMin(totalMin)}</b><span>Tempo total parado</span></div>
      <div class="kpi"><b>${db.maquinas.filter(m=>m.status==='parada').length}</b><span>Paradas agora</span></div>
    </div>

    <div class="section-head"><h2>Pareto por categoria</h2></div>
    <div id="pareto"></div>

    <div class="section-head"><h2>Top 5 — máquinas com mais paradas</h2></div>
    <div id="topMaq" style="display:flex; flex-direction:column; gap:10px;"></div>

    <div style="margin-top:20px;">
      <button class="btn block primary" onclick="exportarCSV()">⬇️ Exportar CSV</button>
    </div>
  `;

  const pareto = document.getElementById('pareto');
  if(ordenado.length === 0){
    pareto.innerHTML = '<div class="empty">Nenhuma parada encerrada para calcular.</div>';
  } else {
    pareto.innerHTML = ordenado.map(([cat, min])=>{
      const pct = Math.round(min / maxCat * 100);
      return `
        <div class="pareto-row">
          <div class="plabel"><span>${escapeHtml(cat)}</span><span>${fmtDuracaoMin(min)}</span></div>
          <div class="ptrack"><div class="pfill" style="width:${pct}%"></div></div>
        </div>
      `;
    }).join('');
  }

  const tm = document.getElementById('topMaq');
  if(topMaq.length === 0){
    tm.innerHTML = '<div class="empty">Sem dados suficientes.</div>';
  } else {
    tm.innerHTML = topMaq.map(([nome, cnt], i)=>`
      <div style="background:var(--panel); border:1px solid var(--border); border-radius:10px; padding:11px 14px; display:flex; align-items:center; gap:12px; box-shadow:var(--shadow);">
        <span style="background:var(--red-dim); color:var(--red); width:26px; height:26px; border-radius:50%; font-weight:900; font-size:12px; display:flex; align-items:center; justify-content:center; flex-shrink:0;">${i+1}</span>
        <span style="flex:1; font-weight:700; font-size:14px;">${escapeHtml(nome)}</span>
        <span style="font-family:var(--mono); font-weight:800; color:var(--red);">${cnt}</span>
      </div>
    `).join('');
  }
}

function exportarCSV(){
  const encerradas = db.paradas.filter(p=>p.status === 'encerrada');
  if(encerradas.length === 0){ toast('Nada para exportar ainda.','red'); return; }
  const headers = ['Data','Nº','Máquina','Setor','Turno','Duração (min)','Impacto','Tipo de Falha','Subcausa','Componente','Causa Raiz','Ação no Componente','Ação Preventiva','Responsável','Observação'];
  const rows = encerradas.map(p=>{
    const m = getMaquina(p.maquinaId);
    return [
      new Date(p.horaInicio).toLocaleDateString('pt-BR'),
      p.numero, m?.nome||'', p.setor, p.turno, p.duracaoMin||'',
      p.impacto, p.categoria, p.subcausa, p.componente, p.causaRaiz,
      p.acaoComponente, p.acaoPreventiva, p.responsavel, p.observacao
    ];
  });
  const csv = [headers, ...rows]
    .map(r=>r.map(v=>`"${String(v??'').replace(/"/g,'""')}"`).join(';'))
    .join('\n');
  const blob = new Blob(['\uFEFF'+csv], {type:'text/csv;charset=utf-8;'});
  const url = URL.createObjectURL(blob);
  const a = document.createElement('a');
  a.href = url;
  a.download = `paradas_seara_${new Date().toISOString().slice(0,10)}.csv`;
  a.click();
  URL.revokeObjectURL(url);
  toast('CSV exportado','green');
}

async function abrirQR(){
  if(!window.jsQR){ toast('Leitor QR não carregou. Digite o nome.','red'); return; }
  document.getElementById('qrOverlay').classList.add('ativo');
  document.getElementById('qrHint').textContent = 'Procurando código…';
  try{
    qrStream = await navigator.mediaDevices.getUserMedia({video:{facingMode:'environment'}});
    const video = document.getElementById('qrVideo');
    video.srcObject = qrStream;
    await video.play();
    const canvas = document.createElement('canvas');
    const ctx = canvas.getContext('2d');
    function loop(){
      if(video.readyState === video.HAVE_ENOUGH_DATA){
        canvas.width = video.videoWidth;
        canvas.height = video.videoHeight;
        ctx.drawImage(video, 0, 0, canvas.width, canvas.height);
        const img = ctx.getImageData(0, 0, canvas.width, canvas.height);
        const code = window.jsQR(img.data, img.width, img.height);
        if(code && code.data){
          const codigo = code.data.trim();
          // Tenta achar máquina por QR ou nome
          let maq = db.maquinas.find(m=>m.qr === codigo || m.nome.toLowerCase() === codigo.toLowerCase());
          if(!maq){
            maq = {id:'m'+Date.now(), nome:codigo, setor:'', prioridade:2, status:'parada', qr:codigo};
            db.maquinas.push(maq);
            salvarDB();
          }
          const f = document.getElementById('fMaquina');
          if(f) f.value = maq.nome;
          const s = document.getElementById('fSetor');
          if(s && maq.setor) s.value = maq.setor;
          document.getElementById('qrHint').textContent = '✅ '+codigo;
          if(navigator.vibrate) navigator.vibrate(50);
          toast('Máquina identificada: '+maq.nome,'green');
          setTimeout(fecharQR, 500);
          return;
        }
      }
      qrLoopId = requestAnimationFrame(loop);
    }
    loop();
  }catch(err){
    document.getElementById('qrHint').textContent = 'Câmera indisponível — digite manualmente.';
    toast('Não foi possível acessar a câmera.','red');
  }
}
function fecharQR(){
  document.getElementById('qrOverlay').classList.remove('ativo');
  if(qrLoopId) cancelAnimationFrame(qrLoopId);
  if(qrStream){ qrStream.getTracks().forEach(t=>t.stop()); qrStream = null; }
}

function toggleDitado(){
  const SR = window.SpeechRecognition || window.webkitSpeechRecognition;
  if(!SR){ toast('Ditado não suportado neste navegador.','red'); return; }
  const btn = document.getElementById('btnMic');
  if(ditando){ reconhecimento?.stop(); return; }
  reconhecimento = new SR();
  reconhecimento.lang = 'pt-BR';
  reconhecimento.continuous = true;
  reconhecimento.interimResults = true;
  const base = document.getElementById('fCausaRaiz').value;
  reconhecimento.onstart = ()=>{ ditando = true; btn.classList.add('rec'); toast('🎤 Ouvindo…'); };
  reconhecimento.onresult = ev=>{
    let txt = '';
    for(let i=0; i<ev.results.length; i++) txt += ev.results[i][0].transcript;
    document.getElementById('fCausaRaiz').value = (base ? base+' ' : '') + txt;
  };
  reconhecimento.onerror = ()=>{};
  reconhecimento.onend = ()=>{ ditando = false; btn?.classList.remove('rec'); };
  reconhecimento.start();
}
